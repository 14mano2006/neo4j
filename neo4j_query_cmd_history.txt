============================================================
1. CREATE CONSTRAINTS
============================================================

CREATE CONSTRAINT student_id_unique IF NOT EXISTS
FOR (s:Student) REQUIRE s.student_id IS UNIQUE;

CREATE CONSTRAINT faculty_id_unique IF NOT EXISTS
FOR (f:Faculty) REQUIRE f.faculty_id IS UNIQUE;

CREATE CONSTRAINT course_id_unique IF NOT EXISTS
FOR (c:Course) REQUIRE c.course_id IS UNIQUE;

CREATE CONSTRAINT department_id_unique IF NOT EXISTS
FOR (d:Department) REQUIRE d.department_id IS UNIQUE;

============================================================
10. VERIFY NODE COUNTS
============================================================

MATCH (n)
RETURN labels(n) AS NodeType, count(n) AS TotalNodes
ORDER BY NodeType;


============================================================
11. VERIFY RELATIONSHIP COUNTS
============================================================

MATCH ()-[r]->()
RETURN type(r) AS RelationshipType, count(r) AS TotalRelationships
ORDER BY RelationshipType;


============================================================
12. DISPLAY THE COMPLETE GRAPH
============================================================

MATCH (n)
OPTIONAL MATCH (n)-[r]-(m)
RETURN n, r, m;


============================================================
13. QUERY 1: DISPLAY ALL STUDENTS
============================================================

MATCH (s:Student)
RETURN s.student_id AS StudentID, s.name AS StudentName, s.semester AS Semester
ORDER BY StudentName;


============================================================
14. QUERY 2: DISPLAY ALL COURSES
============================================================

MATCH (c:Course)
RETURN c.course_id AS CourseID, c.name AS CourseName,
       c.credits AS Credits, c.semester AS Semester
ORDER BY CourseName;


============================================================
15. QUERY 3: FIND STUDENTS ENROLLED IN GRAPH ANALYTICS
============================================================

MATCH (s:Student)-[:ENROLLED_IN]->(c:Course {name: 'Graph Analytics'})
RETURN s.name AS StudentName
ORDER BY StudentName;


============================================================
16. QUERY 4: FIND COURSES WITH MORE THAN 3 CREDITS
============================================================

MATCH (c:Course)
WHERE c.credits > 3
RETURN c.name AS CourseName, c.credits AS Credits
ORDER BY CourseName;


============================================================
17. QUERY 5: FIND COURSES TAUGHT BY DR. MEERA SEN
============================================================

MATCH (f:Faculty {name: 'Dr. Meera Sen'})-[:TEACHES]->(c:Course)
RETURN f.name AS FacultyName, c.name AS CourseName
ORDER BY CourseName;


============================================================
18. QUERY 6: FIND ALL STUDENTS IN SEMESTER 5
============================================================

MATCH (s:Student)
WHERE s.semester = 5
RETURN s.name AS StudentName, s.semester AS Semester
ORDER BY StudentName;


============================================================
19. QUERY 7: FIND STUDENTS BELONGING TO COMPUTER SCIENCE
============================================================

MATCH (s:Student)-[:BELONGS_TO]->(d:Department {name: 'Computer Science'})
RETURN s.name AS StudentName, d.name AS Department
ORDER BY StudentName;


============================================================
20. QUERY 8: FIND COURSES OFFERED BY INFORMATION TECHNOLOGY
============================================================

MATCH (d:Department {name: 'Information Technology'})-[:OFFERS]->(c:Course)
RETURN d.name AS Department, c.name AS CourseName
ORDER BY CourseName;


============================================================
21. QUERY 9: COUNT STUDENTS ENROLLED IN EACH COURSE
============================================================

MATCH (s:Student)-[:ENROLLED_IN]->(c:Course)
RETURN c.name AS CourseName, count(s) AS StudentCount
ORDER BY CourseName;


============================================================
22. QUERY 10: FIND COURSES TAKEN BY AARAV SHARMA
============================================================

MATCH (s:Student {name: 'Aarav Sharma'})-[:ENROLLED_IN]->(c:Course)
RETURN s.name AS StudentName, c.name AS CourseName
ORDER BY CourseName;


============================================================
23. QUERY 11: FIND STUDENTS ENROLLED IN NOSQL DATABASES
============================================================

MATCH (s:Student)-[:ENROLLED_IN]->(c:Course {name: 'NoSQL Databases'})
RETURN s.name AS StudentName
ORDER BY StudentName;


============================================================
24. QUERY 12: FIND FACULTY AND THE COURSES THEY TEACH
============================================================

MATCH (f:Faculty)-[:TEACHES]->(c:Course)
RETURN f.name AS FacultyName, collect(c.name) AS CoursesTaught
ORDER BY FacultyName;


============================================================
25. QUERY 13: COUNT STUDENTS IN EACH DEPARTMENT
============================================================

MATCH (s:Student)-[:BELONGS_TO]->(d:Department)
RETURN d.name AS Department, count(s) AS StudentCount
ORDER BY Department;


============================================================
26. QUERY 14: FIND STUDENTS ENROLLED IN TWO OR MORE COURSES
============================================================

MATCH (s:Student)-[:ENROLLED_IN]->(c:Course)
WITH s, count(c) AS NumberOfCourses
WHERE NumberOfCourses >= 2
RETURN s.name AS StudentName, NumberOfCourses
ORDER BY StudentName;


============================================================
